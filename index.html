function pad(n){ return String(n).padStart(2,'0'); }

function fmtICSDate(d){

return d.getUTCFullYear()+pad(d.getUTCMonth()+1)+pad(d.getUTCDate())+'T'+pad(d.getUTCHours())+pad(d.getUTCMinutes())+'00Z';

}

function buildICS(entry){

const start = new Date(entry.when);

const end = new Date(start.getTime() + (entry.dauer||30)*60000);

const lines = [

'BEGIN:VCALENDAR','VERSION:2.0','PRODID:-//Trainingsauswahl//DE','BEGIN:VEVENT',

'UID:'+entry.id+'@trainingsauswahl',

'DTSTAMP:'+fmtICSDate(new Date()),

'DTSTART:'+fmtICSDate(start),

'DTEND:'+fmtICSDate(end),

'SUMMARY:'+entry.name,

'DESCRIPTION:'+entry.art+' u00b7 '+entry.quelle+' u00b7 '+entry.dauer+' Min',

'END:VEVENT','END:VCALENDAR'

];

return lines.join('rn');

}

function downloadICS(entry){

const blob = new Blob([buildICS(entry)], {type:'text/calendar'});

const url = URL.createObjectURL(blob);

const a = document.createElement('a');

a.href = url;

a.download = entry.name.replace(/[^a-z0-9]+/gi,'_').toLowerCase() + '.ics';

document.body.appendChild(a); a.click(); document.body.removeChild(a);

URL.revokeObjectURL(url);

}

el.innerHTML = '<div class="empty">Noch nichts geplant. Wähle oben ein Workout aus und plane einen Termin.</div>';

const item = document.createElement('div');

item.querySelector('[data-act="del"]').onclick = async ()=>{

async function loadWorkouts(){

async function saveWorkouts(){

}


function renderChips(containerId, options, key){

const el = document.getElementById(containerId);

el.innerHTML = '';

options.forEach(opt=>{

const chip = document.createElement('div');

chip.className = 'chip' + (filters[key].has(opt) ? ' active' : '');

chip.textContent = opt;

chip.onclick = ()=>{

if(filters[key].has(opt)) filters[key].delete(opt);

else filters[key].add(opt);

renderChips(containerId, options, key);

updateMatchCount();

};

el.appendChild(chip);

});

}


function renderTrainerSelect(){

const sel = document.getElementById('trainer-select');

const trainers = [...new Set(workouts.map(w=>w.trainer).filter(Boolean))].sort();

sel.innerHTML = '<option value="">Alle</option>' + trainers.map(t=>`<option value="${t}">${t}</option>`).join('');

sel.value = filters.trainer;

}


function matches(w){

if(filters.art.size && !filters.art.has(w.art)) return false;

if(filters.quelle.size && !filters.quelle.has(w.quelle)) return false;

if(filters.intensitaet.size && !filters.intensitaet.has(w.intensitaet)) return false;

if(filters.min!==null && !isNaN(filters.min) && w.dauer < filters.min) return false;

if(filters.max!==null && !isNaN(filters.max) && w.dauer > filters.max) return false;

if(filters.trainer && w.trainer !== filters.trainer) return false;

return true;

}


function updateMatchCount(){

const n = workouts.filter(matches).length;

document.getElementById('match-count').textContent =

n === 0 ? 'Keine Treffer mit diesen Filtern' :

n === 1 ? '1 Workout passt' : n + ' Workouts passen';

renderList();

}


function pickRandom(){

const pool = workouts.filter(matches);

if(pool.length === 0){

resultEl.classList.remove('show');

return;

const linkEl = document.getElementById('open-link');

if(w.link){ linkEl.href = w.link; linkEl.style.display='inline-flex'; }

else{ linkEl.style.display='none'; }

resultEl.classList.add('show');

}

}
